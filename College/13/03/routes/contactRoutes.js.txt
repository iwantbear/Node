const express = require("express");
const router = express.Router();

const cookieParser = require("cookie-parser");
const checkLogin = require("../middlewares/checkLogin");

const {
  getAllContacts,
  createContact, 
  getContact,
  updateContact,
  deleteContact,
  addContactForm,
} = require("../controllers/contactController4mongo");
//} = require("../controllers/contactController4mysql");

router.use(cookieParser());

// http://localhost:3000/contacts/3
router.route("/").get(getAllContacts);
router.route("/add").get(checkLogin, addContactForm).post(checkLogin, createContact);  // GET: contactRoutes-4.js POST: contactRoutes-5.js
router.route("/:id").get(checkLogin, getContact).put(checkLogin, updateContact).delete(checkLogin, deleteContact);

module.exports = router;
