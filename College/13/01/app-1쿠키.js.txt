const express = require("express");
const app = express();
const dbConnect = require("./config/dbConnect4mongo");
const methodOverride = require("method-override"); // app-15.js   npm i method-override
const cookieParser = require("cookie-parser"); // cookie

const port = 3000;  
dbConnect();

app.set("view engine", "ejs");
app.set("views", "./views");
app.engine('html', require('ejs').renderFile); 
app.use(express.static("./public"));
app.use(methodOverride("_method")); // app-15.js   npm i method-override
app.use(cookieParser()); // cookie

//app.get("/", (req, res) => {
//  res.status(200).send("Hello Node!");
//});

// cookie
app.get("/", (req, res) => {
  res.cookie("Kim", "1234", { httpOnly: true });
  res.send("쿠키 생성");
});

app.get("/cookie", (req, res) => {
  console.log(req.cookies);
});

app.get("/delete-cookie", (req, res) => {
  res.clearCookie("Kim");
  res.send("쿠키 삭제");
});
// cookie
 
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// http://localhost:3000/
// app.use("/", require("./routes/loginRoutes"));
// http://localhost:3000/contacts/3
app.use("/contacts", require("./routes/contactRoutes"));


app.listen(port, () => {
  console.log(`${port}번 포트에서 서버 실행 중`);
});
